cmake_minimum_required(VERSION 3.28)
project(NdiMachine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows x64 / MSVC required")
endif()
set(DECKLINK_SDK "${CMAKE_CURRENT_SOURCE_DIR}/.deps/Blackmagic DeckLink SDK 16.0" CACHE PATH "Installed Desktop Video SDK 16.0")
if(NOT EXISTS "${DECKLINK_SDK}/Win/include/DeckLinkAPI.idl")
  message(FATAL_ERROR "Set DECKLINK_SDK to the local SDK 16.0 installation. No simulation fallback.")
endif()
file(STRINGS "${DECKLINK_SDK}/Win/include/DeckLinkAPIVersion.h" DECKLINK_VERSION_LINE REGEX "^#define BLACKMAGIC_DECKLINK_API_VERSION_STRING")
if(NOT DECKLINK_VERSION_LINE MATCHES "\"16\\.0\"")
  message(FATAL_ERROR "This project is pinned to SDK 16.0. Review dependency changes explicitly.")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/decklink")
find_program(MIDL midl REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/decklink/DeckLinkAPI.h" "${CMAKE_CURRENT_BINARY_DIR}/decklink/DeckLinkAPI_i.c"
 COMMAND "${MIDL}" /nologo /env x64 /I "${DECKLINK_SDK}/Win/include" /h DeckLinkAPI.h /iid DeckLinkAPI_i.c /out "${CMAKE_CURRENT_BINARY_DIR}/decklink" "${DECKLINK_SDK}/Win/include/DeckLinkAPI.idl"
 DEPENDS "${DECKLINK_SDK}/Win/include/DeckLinkAPI.idl" VERBATIM)
add_library(decklink_ids STATIC "${CMAKE_CURRENT_BINARY_DIR}/decklink/DeckLinkAPI_i.c")
target_include_directories(decklink_ids PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/decklink")
# Milestone A: only implemented targets are exposed. Media/GUI targets follow in later milestones.
add_executable(decklink-sdk-check native/sdk_check.cpp)
target_include_directories(decklink-sdk-check PRIVATE "${DECKLINK_SDK}/Win/include")
target_compile_options(decklink-sdk-check PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(decklink-sdk-check PRIVATE decklink_ids ole32 oleaut32)
enable_testing()
add_test(NAME sdk-read-only-inventory COMMAND decklink-sdk-check)
add_executable(core-tests tests/core_tests.cpp)
target_include_directories(core-tests PRIVATE native)
target_compile_options(core-tests PRIVATE /W4 /EHsc /utf-8)
add_test(NAME media-core COMMAND core-tests)
set(NDI_SDK "C:/Program Files/NDI/NDI 6 SDK" CACHE PATH "Licensed standard NDI SDK 6.3.1")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
 URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
add_executable(ndi-probe native/ndi_probe.cpp)
target_include_directories(ndi-probe PRIVATE "${NDI_SDK}/Include")
target_compile_options(ndi-probe PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(ndi-probe PRIVATE nlohmann_json::nlohmann_json "${NDI_SDK}/Lib/x64/Processing.NDI.Lib.x64.lib")
add_executable(ndi-sdi-proof native/sdi_proof.cpp)
target_include_directories(ndi-sdi-proof PRIVATE native "${NDI_SDK}/Include" "${DECKLINK_SDK}/Win/include")
target_compile_options(ndi-sdi-proof PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(ndi-sdi-proof PRIVATE decklink_ids nlohmann_json::nlohmann_json "${NDI_SDK}/Lib/x64/Processing.NDI.Lib.x64.lib" ole32 oleaut32 gdi32 avrt)

add_executable(audio-host native/audio_host.cpp)
target_include_directories(audio-host PRIVATE native)
target_compile_options(audio-host PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(audio-host PRIVATE nlohmann_json::nlohmann_json ole32 uuid avrt advapi32)
add_executable(audio-tests tests/audio_tests.cpp)
target_include_directories(audio-tests PRIVATE native)
target_compile_options(audio-tests PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(audio-tests PRIVATE ole32)
add_test(NAME audio-bridge COMMAND audio-tests)
add_executable(media-worker native/media_worker.cpp)
target_include_directories(media-worker PRIVATE native "${NDI_SDK}/Include" "${DECKLINK_SDK}/Win/include")
target_compile_options(media-worker PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(media-worker PRIVATE decklink_ids nlohmann_json::nlohmann_json "${NDI_SDK}/Lib/x64/Processing.NDI.Lib.x64.lib" ole32 oleaut32 gdi32 avrt advapi32)
